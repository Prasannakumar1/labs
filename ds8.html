#include <stdio.h>
#include <stdlib.h>

struct node
{
    int ssn, sal;
    char name[20], dept[10], desg[20], phno[15];
    struct node *next;
    struct node *prev;
};

typedef struct node *NODE;
NODE first, last, temp, cur;
int count = 0;

void create()
{
    temp = (struct node *)malloc(sizeof(struct node));
    temp->next = NULL;
    temp->prev = NULL;
    printf("\n Enter the SSN:");
    scanf("%d", &temp->ssn);
    printf("\n Enter the Name:");
    scanf("%s", temp->name);
    printf("\n Enter the dept:");
    scanf("%s", temp->dept);
    printf("\n Enter the designation:");
    scanf("%s", temp->desg);
    printf("\n Enter the salary:");
    scanf("%d", &temp->sal);
    printf("\n Enter the Contact Number:");
    scanf("%s", temp->phno);
}

void insertAtFrontDLL()
{
    create();
    if (first == NULL)
    {
        first = last = temp;
    }
    else
    {
        temp->next = first;
        first->prev = temp;
        first = temp;
    }
    count++;
    printf(" No of employees = %d ", count);
}

void insertAtRearDLL()
{
    create();
    if (first == NULL)
    {
        first = last = temp;
    }
    else
    {
        last->next = temp;
        temp->prev = last;
        last = temp;
    }
    count++;
    printf(" No of employees = %d ", count);
}

void deleteAtFrontDLL()
{
    if (first == NULL)
    {
        printf("List is empty");
    }
    else if (first == last)
    {
        printf("The record is deleted with SSN: %d\n", first->ssn);
        free(first);
        count--;
        printf("No of employees = %d ", count);
        first = last = NULL;
    }
    else
    {
        cur = first;
        first = first->next;
        first->prev = NULL;
        printf("The record is deleted with SSN: %d\n", cur->ssn);
        free(cur);
        count--;
        printf("No of employees = %d ", count);
    }
}

void deleteAtRearDLL()
{
    if (first == NULL)
    {
        printf("List empty");
    }
    else if (last == first)
    {
        printf("The record is deleted with SSN: %d\n", last->ssn);
        free(last);
        count--;
        printf("No of employees = %d ", count);
        last = first = NULL;
    }
    else
    {
        cur = last;
        last = last->prev;
        last->next = NULL;
        printf("The record is deleted with SSN: %d\n", cur->ssn);
        free(cur);
        count--;
        printf("No of employees = %d ", count);
    }
}

void displayDLL()
{
    if (first == NULL)
    {
        printf("List is empty");
    }
    else
    {
        printf("\n The contents of the list are\n");
        printf("|--SSN--|--NAME--|--DEPT--|-DESIGNATION--|--SALARY--|--PHONE NO--\n");
        cur = first;
        while (cur != NULL)
        {
            printf("%d\t%s\t%s\t%s\t%d\t%s\n", cur->ssn, cur->name, cur->dept, cur->desg, cur->sal, cur->phno);
            cur = cur->next;
        }
        printf("No of employees = %d ", count);
    }
}

void main()
{
    int ch, n, i;
    first = last = NULL;
    for (;;)
    {
        printf("\n--------MENU-------------------\n");
        printf("\n1.Create DLL with n Employees");
