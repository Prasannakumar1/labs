#include <stdio.h>
#include <conio.h>
#include <math.h>

#define MAX 20
int top = -1;
float stack[MAX];
char postfix[MAX];

void push(float);
float pop();
int isoperand(char);
float operate(float, float, char);

void main()
{
    int i = 0;
    float ans, op1, op2;
    printf("Enter Expression:");
    scanf("%s", postfix);

    while (postfix[i] != '\0')
    {
        if (isoperand(postfix[i]))
            push(postfix[i] - 48);
        else
        {
            op1 = pop();
            op2 = pop();
            ans = operate(op1, op2, postfix[i]);
            push(ans);
            printf("%f %c %f = %f\n", op2, postfix[i], op1, ans);
        }
        i++;
    }

    printf("%f", stack[top]);
}

int isoperand(char x)
{
    if (x >= '0' && x <= '9')
        return 1;
    else
        return 0;
}

void push(float x)
{
    stack[++top] = x;
}

float pop()
{
    return stack[top--];
}

float operate(float op1, float op2, char a)
{
    switch (a)
    {
    case '+':
        return op2 + op1;
    case '-':
        return op2 - op1;
    case '*':
        return op2 * op1;
    case '/':
        return op2 / op1;
    case '^':
        return pow(op2, op1);
    }
}
