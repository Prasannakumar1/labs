import java.util.Arrays; public class BellmanFordRouting { static class Edge { int src, dest, weight; Edge(int s, int d, int w) { src = s; dest = d; weight = w; } } public static void bellmanFord(int V, int E, Edge[] edges, int sourceNode) { int[] dist = new int[V]; int[] parent = new int[V]; // Step 1: Initialize distances from source to all other vertices as INFINITY Arrays.fill(dist, 99999); Arrays.fill(parent, -1); dist[sourceNode] = 0; // Step 2: Relax all edges |V| - 1 times (Distance Vector Update) for (int i = 1; i < V; ++i) { for (int j = 0; j < E; ++j) { int u = edges[j].src; int v = edges[j].dest; int weight = edges[j].weight; if (dist[u] != 99999 && dist[u] + weight < dist[v]) { dist[v] = dist[u] + weight; parent[v] = u; } } } // Step 3: Check for negative-weight cycles for (int j = 0; j < E; ++j) { int u = edges[j].src; int v = edges[j].dest; int weight = edges[j].weight; if (dist[u] != 99999 && dist[u] + weight < dist[v]) { System.out.println("Graph contains negative weight cycle!"); return; } } // Output Final Path Vector Routing Table System.out.println("\n=== FINAL ROUTING TABLE (From Node " + sourceNode + ") ==="); for (int i = 0; i < V; ++i) { System.out.println("Node " + sourceNode + " -> Node " + i + " | Min Cost: " + dist[i]); } } }
