import java.util.Random; import java.util.Scanner; public class SlidingWindow { public static void main(String[] args) { int totalFrames = 8; int windowSize = 4; int sent = 0; int ack = 0; Random random = new Random(); System.out.println("=== STARTING SLIDING WINDOW SIMULATION ==="); // Main Loop: Continue until all frames are ACKed while (ack < totalFrames) { // 1. TRANSMIT WINDOW for (int i = sent; i < ack + windowSize && i < totalFrames; i++) { System.out.println("[SENDER] Transmitted Frame " + i); sent++; } // 2. RECEIVE ACKNOWLEDGMENTS for (int i = ack; i < sent; i++) { boolean isLost = random.nextInt(10) < 2; // Simulated Loss if (!isLost) { System.out.println("[RECEIVER] Frame " + i + " OK -> Sent ACK " + (i + 1)); ack++; } else { System.out.println("[CHANNEL] ❌ Frame/ACK " + i + " LOST!"); System.out.println("[SENDER] ⏰ Timeout! Resending from Frame " + ack); sent = ack; // Go-Back-N Window Reset break; } } } System.out.println("🎉 TRANSMISSION COMPLETE!"); } }
