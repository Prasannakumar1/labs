import java.util.Scanner; public class LeakyBucket { public static void main(String[] args) { int bucketCapacity = 10; // Maximum buffer size int outputRate = 3; // Constant leak rate per second int[] incomingPackets = {4, 7, 2, 8, 1}; // Packet bursts per time unit int storedBuffer = 0; System.out.println("=== LEAKY BUCKET CONGESTION CONTROL SIMULATION ==="); System.out.println("Bucket Capacity: " + bucketCapacity + " | Leak Rate: " + outputRate + " units/sec\n"); // Process incoming burst packets over time for (int t = 0; t < incomingPackets.length; t++) { int packet = incomingPackets[t]; System.out.println("--- SECOND " + (t + 1) + ": Incoming Packet Size = " + packet + " ---"); // Check if bucket can accept incoming packet without overflowing if (storedBuffer + packet <= bucketCapacity) { storedBuffer += packet; System.out.println("[ACCEPT] Added " + packet + " units. Buffer Status: " + storedBuffer + "/" + bucketCapacity); } else { int dropped = (storedBuffer + packet) - bucketCapacity; storedBuffer = bucketCapacity; System.out.println("[OVERFLOW] ❌ Buffer full! " + dropped + " units DROPPED. Buffer Status: " + storedBuffer + "/" + bucketCapacity); } // Leak packets at constant output rate int leaked = Math.min(storedBuffer, outputRate); storedBuffer -= leaked; System.out.println("[LEAK] 💧 Transmitted " + leaked + " units. Remaining Buffer: " + storedBuffer); System.out.println(); } // Empty remaining buffer after inputs finish while (storedBuffer > 0) { int leaked = Math.min(storedBuffer, outputRate); storedBuffer -= leaked; System.out.println("[EMPTYING BUFFER] 💧 Transmitted " + leaked + " units. Remaining Buffer: " + storedBuffer); } System.out.println("\n🎉 CONGESTION CONTROL COMPLETE: Bucket Empty!"); } }
