import java.math.BigInteger; import java.util.Scanner; public class RSAAlgorithm { public static void main(String[] args) { // Step 1: Select two prime numbers (p and q) BigInteger p = BigInteger.valueOf(61); BigInteger q = BigInteger.valueOf(53); // Step 2: Compute n = p * q BigInteger n = p.multiply(q); // Step 3: Compute Euler's Totient phi(n) = (p-1)*(q-1) BigInteger phi = (p.subtract(BigInteger.ONE)).multiply(q.subtract(BigInteger.ONE)); // Step 4: Choose public key exponent 'e' such that 1 < e < phi and gcd(e, phi) = 1 BigInteger e = BigInteger.valueOf(17); // Step 5: Calculate private key exponent 'd' such that (d * e) % phi = 1 BigInteger d = e.modInverse(phi); System.out.println("Public Key (e, n) : (" + e + ", " + n + ")"); System.out.println("Private Key (d, n): (" + d + ", " + n + ")"); // Input message string String message = "HELLO"; System.out.println("\nOriginal Message: " + message); // Step 6: Encryption: Cipher = (Message^e) % n byte[] bytes = message.getBytes(); BigInteger[] encrypted = new BigInteger[bytes.length]; for (int i = 0; i < bytes.length; i++) { BigInteger msgVal = BigInteger.valueOf(bytes[i]); encrypted[i] = msgVal.modPow(e, n); } System.out.println("Encrypted Ciphertext: " + java.util.Arrays.toString(encrypted)); // Step 7: Decryption: Plaintext = (Cipher^d) % n byte[] decryptedBytes = new byte[encrypted.length]; for (int i = 0; i < encrypted.length; i++) { BigInteger decryptedVal = encrypted[i].modPow(d, n); decryptedBytes[i] = decryptedVal.byteValue(); } String decryptedMessage = new String(decryptedBytes); System.out.println("Decrypted Message : " + decryptedMessage); } }
