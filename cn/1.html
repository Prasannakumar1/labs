NS2 Lab: 3-Node Network (Duplex Links, Queue & Drop Analysis)

NS2 Tcl Script & AWK Drop Analyzer
# --- 1. NS2 Tcl Script (network.tcl) ---
set ns [new Simulator]

set tf [open trace.tr w]
$ns trace-all $tf

set n0 [$ns node]
set n1 [$ns node]
set n2 [$ns node]

# Duplex link configurations (Bottleneck at n1-n2)
$ns duplex-link $n0 $n1 1Mb 10ms DropTail
$ns duplex-link $n1 $n2 0.2Mb 10ms DropTail

# Queue Limit setup
$ns queue-limit $n1 $n2 5

set udp [new Agent/UDP]
$ns attach-agent $n0 $udp
set null [new Agent/Null]
$ns attach-agent $n2 $null
$ns connect $udp $null

set cbr [new Application/Traffic/CBR]
$cbr set packetSize_ 500
$cbr set interval_ 0.005
$cbr attach-agent $udp

$ns at 0.1 "$cbr start"
$ns at 5.0 "$ns halt"
$ns run

# --- 2. AWK Script (droppackets.awk) ---
BEGIN {
    drop_count = 0;
}
{
    if ($1 == "d" && $5 == "RTR") {
        drop_count++;
    }
}
END {
    printf("Total Packets Dropped = %d\n", drop_count);
}
