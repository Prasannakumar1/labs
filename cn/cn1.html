# --- 1. NS2 Tcl Script (network.tcl) ---
set ns [new Simulator]

$ns color 1 Blue
$ns color 2 Red

set nf [open out.nam w]
$ns namtrace-all $nf
set tf [open trace.tr w]
$ns trace-all $tf

proc finish {} {
    global ns nf tf
    $ns flush-trace
    close $nf
    close $tf
    exec nam out.nam &
    exit 0
}

set n0 [$ns node]
set n1 [$ns node]
set n2 [$ns node]

# Duplex link configurations
$ns duplex-link $n0 $n1 1Mb 10ms DropTail
$ns duplex-link $n1 $n2 0.2Mb 10ms DropTail

# Bottleneck Queue limit setup
$ns queue-limit $n1 $n2 5

set udp [new Agent/UDP]
$udp set fid_ 2
$ns attach-agent $n0 $udp
set null [new Agent/Null]
$ns attach-agent $n2 $null
$ns connect $udp $null

set cbr [new Application/Traffic/CBR]
$cbr set packetSize_ 500
$cbr set interval_ 0.005
$cbr attach-agent $udp

$ns at 0.1 "$cbr start"
$ns at 4.5 "$cbr stop"
$ns at 5.0 "finish"
$ns run

# --- 2. AWK Drop Calculator (droppackets.awk) ---
BEGIN {
    drop_count = 0;
}
{
    action = $1;
    if (action == "d") {
        drop_count++;
    }
}
END {
    printf("Total dropped = %d\n", drop_count);
}
