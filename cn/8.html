import java.io.*; import java.net.*; public class UDPServer { public static void main(String[] args) throws IOException { // 1. Create DatagramSocket at Server side DatagramSocket serverSocket = new DatagramSocket(); InetAddress clientAddress = InetAddress.getByName("127.0.0.1"); int clientPort = 5000; BufferedReader reader = new BufferedReader(new InputStreamReader(System.in)); System.out.println("UDP Server Ready. Type message for Client:"); System.out.print("Server Message > "); String message = reader.readLine(); // 2. Convert message string into byte array byte[] sendBuffer = message.getBytes(); // 3. Construct DatagramPacket for Client at Port 5000 DatagramPacket sendPacket = new DatagramPacket( sendBuffer, sendBuffer.length, clientAddress, clientPort ); // 4. Send packet over UDP DatagramSocket serverSocket.send(sendPacket); System.out.println("Datagram message sent to Client."); serverSocket.close(); } }
import java.io.*; import java.net.*; public class UDPClient { public static void main(String[] args) throws IOException { // 1. Bind DatagramSocket to Port 5000 to listen DatagramSocket clientSocket = new DatagramSocket(5000); byte[] receiveBuffer = new byte[1024]; System.out.println("UDP Client Listening on Port 5000..."); // 2. Prepare blank DatagramPacket buffer DatagramPacket receivePacket = new DatagramPacket(receiveBuffer, receiveBuffer.length); // 3. Receive incoming Packet from Server clientSocket.receive(receivePacket); // 4. Convert packet byte array back into String String message = new String(receivePacket.getData(), 0, receivePacket.getLength()); // 5. Display the message typed at Server side System.out.println("[CLIENT DISPLAY] Server Message: " + message); clientSocket.close(); } }
