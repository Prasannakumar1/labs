Client - Server Architecture (Port 4000)
import java.io.*; import java.net.*; public class Server { public static void main(String[] args) throws IOException { // 1. Bind ServerSocket to Port 4000 ServerSocket serverSocket = new ServerSocket(4000); System.out.println("Server listening on port 4000..."); // 2. Accept incoming client connection Socket socket = serverSocket.accept(); System.out.println("Client connected successfully."); // 3. Setup Input stream to receive filename from Client BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream())); String fileName = in.readLine(); System.out.println("Requested file: " + fileName); // 4. Setup Output stream to send file contents to Client PrintWriter out = new PrintWriter(socket.getOutputStream(), true); File file = new File(fileName); if (file.exists() && !file.isDirectory()) { BufferedReader fileReader = new BufferedReader(new FileReader(file)); String line; while ((line = fileReader.readLine()) != null) { out.println(line); // Send content line by line } fileReader.close(); } else { out.println("ERROR: File Not Found on Server!"); } // 5. Close Sockets socket.close(); serverSocket.close(); } }
import java.io.*; import java.net.*; import java.util.Scanner; public class Client { public static void main(String[] args) throws IOException { // 1. Connect to Server running on localhost:4000 Socket socket = new Socket("127.0.0.1", 4000); System.out.println("Connected to TCP Server."); // 2. Read target filename from User Scanner scanner = new Scanner(System.in); System.out.print("Enter file name to request: "); String fileName = scanner.nextLine(); // 3. Send requested filename to Server PrintWriter out = new PrintWriter(socket.getOutputStream(), true); out.println(fileName); // 4. Read file content stream sent back by Server BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream())); String line; System.out.println("\n--- Received File Content ---"); while ((line = in.readLine()) != null) { System.out.println(line); } // 5. Cleanup Resources socket.close(); scanner.close(); } }
