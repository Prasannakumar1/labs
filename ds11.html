#include <stdio.h>
#include <conio.h>
#include <stdlib.h>

void bfs(int n, int a[10][10], int src)
{
    int i, visited[10], q[10], front, rear, u, v;
    for (i = 1; i <= n; i++)
    {
        visited[i] = 0;
    }
    visited[src] = 1;
    front = 0;
    rear = -1;
    q[++rear] = src;

    while (front <= rear)
    {
        u = q[front++];
        for (v = 1; v <= n; v++)
        {
            if (a[u][v] == 1 && visited[v] == 0)
            {
                visited[v] = 1;
                q[++rear] = v;
            }
        }
    }

    for (i = 1; i <= n; i++)
    {
        if (visited[i] == 0)
            printf("%d is not reachable \n", i);
        else
            printf("%d is reachable\n", i);
    }
}

void dfs(int n, int a[10][10], int u, int visited[10])
{
    int v;
    visited[u] = 1;
    for (v = 1; v <= n; v++)
    {
        if (a[u][v] == 1 && visited[v] == 0)
        {
            dfs(n, a, v, visited);
        }
    }
}
