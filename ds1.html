#include <stdio.h>
#include <string.h>
#include <stdlib.h>

struct calendar
{
    char *dayName;
    int date;
    char *description;
} week[7];

void create()
{
    int i;
    for (i = 0; i < 7; i++)
    {
        week[i].dayName = (char *)malloc(20 * sizeof(char));
        week[i].description = (char *)malloc(50 * sizeof(char));
    }
}

void read()
{
    int i;
    for (i = 0; i < 7; i++)
    {
        printf("\n Enter the name of the day: ");
        scanf("%s", week[i].dayName);
        printf(" Enter the date of the day: ");
        scanf("%d", &week[i].date);
        printf(" Enter description of activity: ");
        scanf("%s", week[i].description);
    }
}

void display()
{
    int i;
    for (i = 0; i < 7; i++)
    {
        printf("\n Name of the day: %s", week[i].dayName);
        printf("\n Date of the day: %d", week[i].date);
        printf("\n Activity of the day: %s\n", week[i].description);
    }
}

int main()
{
    create();
    read();
    display();
    return 0;
}
