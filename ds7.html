#include <stdio.h>
#include <stdlib.h>

struct node
{
    char usn[20];
    char name[50];
    char branch[25];
    int sem;
    char phno[13];
    struct node *link;
};

typedef struct node *NODE;
NODE first, temp, cur, prev;

void create()
{
    temp = (struct node *)malloc(sizeof(struct node));
    printf("\n Enter the Name:");
    scanf("%s", temp->name);
    printf("\n Enter the USN:");
    scanf("%s", temp->usn);
    printf("\n Enter the Semester:");
    scanf("%d", &temp->sem);
    printf("\n Enter the Branch:");
    scanf("%s", temp->branch);
    printf("\n Enter the Contact Number:");
    scanf("%s", temp->phno);
    temp->link = NULL;
}

void insertFrontSLL()
{
    create();
    temp->link = first;
    first = temp;
}

void insertRearSLL()
{
    create();
    if (first == NULL)
    {
        first = temp;
        return;
    }
    cur = first;
    while (cur->link != NULL)
        cur = cur->link;
    cur->link = temp;
}

void displaySLL()
{
    int count = 0;
    if (first == NULL)
    {
        printf("****List is empty****\n");
        return;
    }
    printf("\n The contents of the list are\n");
    printf("|----USN----|----Name----|--Branch--|-Sem-|---Phone NO---|\n");
    cur = first;
    while (cur != NULL)
    {
        printf("%s\t%s\t\t%s\t%d\t%s\n", cur->usn, cur->name, cur->branch, cur->sem, cur->phno);
        cur = cur->link;
        count++;
    }
    printf("\nTotal number of student nodes = %d\n", count);
}

void deleteFrontSLL()
{
    if (first == NULL)
    {
        printf("\n ***List is Empty\n***");
        return;
    }
    cur = first;
    first = first->link;
    printf("\n***Record Deleted with USN: %s ***", cur->usn);
    free(cur);
}

void deleteRearSLL()
{
    if (first == NULL)
    {
        printf("\n***List is Empty***");
        return;
    }
    if (first->link == NULL)
    {
        printf("\n***Record Deleted with USN: %s ***", first->usn);
        free(first);
        first = NULL;
        return;
    }
    cur = first;
    while (cur->link != NULL)
    {
        prev = cur;
        cur = cur->link;
    }
    printf("\n***Record deleted with USN: %s ***", cur->usn);
    free(cur);
    prev->link = NULL;
}

void main()
{
    first = NULL;
    int ch;
    for (;;)
    {
        printf("\n-------MAIN MENU-------------\n");
        printf("1: Insert front\n2: Insert Rear\n3: Delete front\n4: Delete Rear\n5: Display\n6: Exit\n");
        printf("Enter your choice----- >");
        scanf("%d", &ch);
        switch (ch)
        {
        case 1:
            insertFrontSLL();
            break;
        case 2:
            insertRearSLL();
            break;
        case 3:
            deleteFrontSLL();
            break;
        case 4:
            deleteRearSLL();
            break;
        case 5:
            displaySLL();
            break;
        case 6:
            exit(0);
        default:
            printf("\nInvalid Choice\n");
        }
    }
}
